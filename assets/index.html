💻 Quick Integration Snippets

Interact with Zev via cURL, Rust, Node.js / npx, or Python.

cURL (/v1/decisions) 
curl -X POST http://127.0.0.1:8080/v1/decisions \
  -H "Content-Type: application/json" \
  -d '{
    "state": "Production database is down with 500 connection refused errors",
    "questions": {
      "route": {
        "type": "choice",
        "instructions": "Route incident to responsible team",
        "options": [
          {"id": "billing", "description": "Payment invoices and billing"},
          {"id": "infra", "description": "Database outages and 500 connection errors"},
          {"id": "sales", "description": "Enterprise sales inquiries"}
        ]
      }
    }
  }'
Rust (zev-rs crate) 
use zev::ZevEngine;

let engine = ZevEngine::default();
let (destination, confidence) = engine.route(
    "Credit card dispute and refund inquiry for invoice #9401",
    [
        ("billing".into(), "Refunds and payment invoices".into()),
        ("tech".into(), "API errors and server issues".into()),
    ].into_iter().collect(),
)?;

println!("Destination: {destination} (Confidence: {confidence:.2})");
Zero-Install CLI (npx) 
npx -y zev-rs route \
  --state "Critical: production database is down with 500 connection refused errors" \
  --routes '{"billing": "Invoice inquiries", "infra": "Database outage and cluster errors"}'
